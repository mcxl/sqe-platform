def check(name, operation, reject=False):
    try:
        operation()
    except ValueError as error:
        if not reject:
            raise
        results.append({'name': name, 'result': 'passed', 'expected': 'rejected', 'actual': str(error)})
    else:
        if reject:
            raise AssertionError(name + ': unsafe input accepted')
        results.append({'name': name, 'result': 'passed', 'expected': 'accepted', 'actual': 'accepted'})

def fixture(arm='B', issue=False, thrown=False):
    calls = [{'arm': arm, 'repetition': '1', 'viewport': v, 'auditName': t,
              'returnedNormally': True, 'issueObserved': False, 'error': None}
             for v in w.VIEWPORTS for t in (w.TYPES if arm == 'B' else ('all',))]
    issues, errors = [], []
    if issue:
        calls[0]['issueObserved'] = True
        issues = [{'arm': arm, 'repetition': '1', 'viewport': 'initial', 'requestedAuditName': calls[0]['auditName'],
                   'auditTypeRawValue': 1, 'auditType': 'XCUIAccessibilityAuditType(rawValue: 1)',
                   'compactDescription': 'Contrast failed', 'detailedDescription': 'Contrast failed for SwiftUI.AccessibilityNode',
                   'element': {'identifier': '', 'label': 'Copy action', 'type': '48',
                               'frame': {'x': 32, 'y': 685.6, 'width': 144, 'height': 58}},
                   'auditStartUptimeSeconds': 10.0, 'callbackElapsedSeconds': 1.0,
                   'callbackEvidence': 'XCUIScreen screenshot observed when the callback ran; not the audit sample rendering'}]
        if thrown:
            calls[0].update(returnedNormally=False, error='Contrast failed')
            errors = [{k: calls[0][k] for k in ('arm', 'repetition', 'viewport', 'auditName', 'error')}]
    completed = ['copy-confirmation'] if thrown else sorted(w.VIEWPORTS)
    complete = {'arm': arm, 'repetition': '1', 'expectedAuditCalls': len(calls), 'attemptedAuditCalls': len(calls),
                'returnedNormallyAuditCalls': len(calls) - int(thrown), 'completedViewports': completed,
                'bothViewportAuditGroupsCompleted': not thrown, 'allAuditTypesAttempted': True,
                'allAuditCallsReturnedNormally': not thrown, 'callbackIssueCount': len(issues),
                'auditErrors': ['Contrast failed'] if thrown else []}
    placements = [] if arm == 'B' else [{'repetition': '1', 'twoFrameReadsIdentical': True,
                'firstFrame': {'x': 10, 'y': 10, 'width': 40, 'height': 40},
                'secondFrame': {'x': 10, 'y': 10, 'width': 40, 'height': 40},
                'viewport': {'x': 0, 'y': 0, 'width': 200, 'height': 200}, 'midpoint': 100}]
    return [arm, 1, calls, issues, [complete], errors, placements]
